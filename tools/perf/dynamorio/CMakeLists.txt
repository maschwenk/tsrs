cmake_minimum_required(VERSION 3.14)
project(tsrs_dr_clients C)
find_package(DynamoRIO REQUIRED)
add_library(icount SHARED icount.c)
configure_DynamoRIO_client(icount)
use_DynamoRIO_extension(icount drmgr)
use_DynamoRIO_extension(icount drreg)
add_library(bbcount SHARED bbcount.c)
configure_DynamoRIO_client(bbcount)
use_DynamoRIO_extension(bbcount drmgr)
use_DynamoRIO_extension(bbcount drreg)
use_DynamoRIO_extension(bbcount drcontainers)
